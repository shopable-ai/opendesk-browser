展开 Sidebar「开发」可运行的现代 Locator 示例
      
async function main() {
  const matches = await page.getByRole('button', {name:'确认记录', exact:true}).count();
  await page.getByTestId('locator-confirm-a').click();
  await page.getByRole('button', {name:'延迟生成目标', exact:true}).click();
  await page.getByTestId('locator-late-target').waitFor({state:'visible', timeout:3000});
  return {
    matches,
    action: await page.locator('#locator-duplicate-result').textContent(),
    delayed: await page.getByTestId('locator-late-target').textContent(),
    readOnly: await page.locator('#locator-readonly-field').getAttribute('readonly')
  };
}
